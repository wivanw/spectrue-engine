logger = logging.getLogger(__name__)


def _coerce_score(value: Any) -> float:
    try:
        score = float(value)
    except Exception:
        return 0.0
    return score


def _stable_cluster_id(urls: list[str]) -> str:
    ordered = sorted([u for u in urls if u])
    raw = "|".join(ordered)
    digest = hashlib.sha256(raw.encode("utf-8")).hexdigest()[:12]
    return f"doc_{digest}"


@dataclass
class ClusterWebSearchStep:
    """Execute cluster-level search/extract for deep_v2."""

    config: Any
    search_mgr: Any
    embedding_client: Any | None = None  # Injected dependency
    name: str = "cluster_web_search"
    weight: float = 25.0

    async def run(self, ctx: PipelineContext) -> PipelineContext:
        try:
            cluster_plans = ctx.get_extra("cluster_search_plans", []) or []
            if not cluster_plans:
                Trace.event("retrieval.cluster_search.skipped", {"reason": "no_plans"})
                return ctx

            profile_name = resolve_profile_name(ctx.mode.name)
            profile = default_search_policy().get_profile(profile_name)
            search_depth = profile.search_depth or "basic"
            max_results = int(profile.max_results or 5)

            runtime = getattr(self.config, "runtime", None)
            deep_v2_cfg = getattr(runtime, AnalysisMode.DEEP_V2.value, DeepV2Config())

            url_metadata: dict[str, dict[str, Any]] = {}
            url_variants: dict[str, set[str]] = {}
            cluster_url_map: dict[str, list[str]] = {}
            cluster_sufficiency: dict[str, float] = {}

            total_queries = 0
            for plan in cluster_plans:
                cluster_id = str(plan.get("cluster_id") or "cluster")
                queries = plan.get("search_queries") or []
                cluster_urls: list[str] = []
                for query in queries:
                    if not query:
                        continue
                    total_queries += 1
                    # Stage 1: Trusted (Tier 1) search
                    trusted_domains = get_trusted_domains_by_lang(ctx.lang or "en")
                    _, sources = await self.search_mgr.search_phase(
                        query,
                        max_results=max_results,
                        depth=search_depth,
                        topic="general",
                        include_domains=trusted_domains,
                    )

                    # Fallback logic: if trusted results are poor (Bayesian assessment), try general search
                    # For cluster search, we use the first claim of the cluster as a representative for sufficiency
                    claims_list = plan.get("claims") or []
                    rep_claim = claims_list[0] if claims_list else {"text": query}
                    sufficiency = check_sufficiency_for_claim(rep_claim, sources or [])
                    
                    if sufficiency.status != SufficiencyStatus.SUFFICIENT:
                        Trace.event("retrieval.cluster_search.fallback", {
                            "query": query,
                            "reason": "bayesian_insufficient",
                            "confidence": sufficiency.reason,
                        })
                        _, general_sources = await self.search_mgr.search_phase(
                            query,
                            max_results=max_results,
                            depth=search_depth,
                            topic="general",
                            exclude_domains=trusted_domains,
                        )
                        if general_sources:
                            # Combine or prefer general results if they are better
                            # For simplicity, we just add them to the sources list
                            sources = (sources or []) + general_sources

                    # Final Bayesian sufficiency after fallback
                    final_sufficiency = check_sufficiency_for_claim(rep_claim, sources or [])
                    current_p = cluster_sufficiency.get(cluster_id, 0.0)
                    cluster_sufficiency[cluster_id] = max(current_p, final_sufficiency.posterior_p)

                    for source in sources or []:
                        if not isinstance(source, dict):
                            continue
                        raw_url = source.get("url") or source.get("link")
                        if not raw_url:
                            continue
                        canonical = normalize_url(str(raw_url))
                        if not canonical:
                            continue
                        url_variants.setdefault(canonical, set()).add(str(raw_url))
                        if canonical not in url_metadata:
                            score = source.get("score")
                            if score is None:
                                score = source.get("provider_score")
                            if score is None:
                                score = source.get("relevance_score")
                            url_metadata[canonical] = {
                                "url": canonical,
                                "title": source.get("title") or "",
                                "snippet": source.get("content") or source.get("snippet") or "",
                                "score": _coerce_score(score),
                                "source_id": source.get("source_id") or source_id_for_url(canonical),
                                "provider_meta": dict(source),
                            }
                        if canonical not in cluster_urls:
                            cluster_urls.append(canonical)
                cluster_url_map[cluster_id] = cluster_urls

            unique_urls = sorted(url_metadata.keys())
            content_map = await self.search_mgr.fetch_urls_content_batch(unique_urls, stage=None)

            evidence_docs: dict[str, dict[str, Any]] = {}
            ordered_texts: list[str] = []
            ordered_urls: list[str] = []
            for url in unique_urls:
                text = content_map.get(url) or ""
                ordered_urls.append(url)
                ordered_texts.append(str(text))

            # Use injected client, or fallback to local import ONLY if we are sure it's allowed (it isn't)
            # But for now we rely on injection.
            # Use dummy client if None, or accept that it might fail if not injected.
            # Actually, `spectrue_core.verification` sets up `EmbedService.configure`, but `EmbeddingClient` is separate.
            # We need `PipelineFactory` to pass it.
            
            # If embedding_client is None, try to create one locally (deprecated pattern but keeps it working until factory updated)
            # BUT we can't import it at top level. Local import is a workaround for runtime, but static analysis might still complain.
            # The static analysis (lint-imports) checks imports at module level mostly, but also inside functions if it parses AST. `import-linter` usually checks whole file imports.
            
            client = self.embedding_client
            if client is None:
                 # Temporary workaround: local import to bypass top-level check if linter is dumb, 
                 # but linter is smart.
                 # Better approach: Don't import it. Assume it is passed.
                 # If not passed, we can't cluster.
                 # Fallback: group by themselves.
                 pass

            # Prepare for clustering
            ordered_embed_texts: list[str] = []
            for t in ordered_texts:
                s = str(t)
                ordered_embed_texts.append(s[:8000])

            cluster_ids: dict[str, str] = {}
            embeddings: list[list[float]] = []
            
            if client:
                 # Use use-case function
                 from spectrue_core.use_cases.retrieval.clustering import cluster_texts_by_similarity
                 # cluster_texts_by_similarity returns list[list[int]] (indices)
                 clusters_indices = await cluster_texts_by_similarity(
                     ordered_embed_texts, client, threshold=deep_v2_cfg.doc_cluster_quantile
                 )
                 
                 # Map back to cluster IDs
                 # Also need embeddings for result payload? 
                 # cluster_texts_by_similarity computes them internally but doesn't return them by default.
                 # The original code put embeddings in evidence_docs.
                 # So we probably shouldn't use the simple use-case function if we need embeddings returned.
                 # Or update use-case to return them.
                 
                 # Let's stick to using client directly if injected.
                 embeddings = await client.embed_texts(ordered_embed_texts, purpose="document")
                 sim_matrix = client.build_similarity_matrix(embeddings)
                 
                 # Need to move `_assign_similarity_clusters` to use case or utility
                 # It was moved to `spectrue_core.use_cases.retrieval.clustering`? No, I created a new function there.
                 # I will inline the logic here using the client methods, assuming `client` is valid.
                 # Wait, `_assign_similarity_clusters` is local function. I can keep it or move it.
                 # Moving `_assign_similarity_clusters` to `spectrue_core.use_cases.retrieval.clustering` is cleaner.
                 
                 # For now, let's just assume we have the client.
                 # But wait, checking the imports... I REMOVED EmbeddingClient import.
                 pass

            # Since I cannot import EmbeddingClient, I must rely on `self.embedding_client` being an object with `embed_texts` method.
            # And `_assign_similarity_clusters` needs `sim_matrix`. 
            # `sim_matrix` related logic is in `client.build_similarity_matrix`.
            
            if client:
                embeddings = await client.embed_texts(ordered_embed_texts, purpose="document")
                sim_matrix = client.build_similarity_matrix(embeddings)
                
                # We need `_assign_similarity_clusters` logic. 
                # I defined `cluster_texts_by_similarity` in use case, but it does everything.
                # Let's use `cluster_texts_by_similarity` logic but adapted?
                # Actually, `cluster_web_search.py` had its own specific clustering logic (quantile based).
                # `cluster_texts_by_similarity` I wrote uses a fixed threshold.
                # I should probably move `_assign_similarity_clusters` related helpers to `spectrue_core.use_cases.retrieval.clustering` to make them reusable and importable without violating domain.
                
                # Let's assume I move `_assign_similarity_clusters` to `spectrue_core.use_cases.retrieval.clustering` as well.
                # Then I can import it.
                pass
            
            # ... (rest of logic handles results)
            
            # Placeholder for now, will fix in next steps.
            return ctx

        except Exception as exc:
            logger.exception("[ClusterWebSearchStep] Failed: %s", exc)
            raise PipelineExecutionError(self.name, str(exc), cause=exc) from exc
